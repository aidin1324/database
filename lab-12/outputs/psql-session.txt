WITH department_averages AS (SELECT department,AVG(salary) AS avg_salary FROM employees GROUP BY department)
SELECT e.first_name,e.salary,e.department,ROUND(d.avg_salary,2) AS department_average
FROM employees e JOIN department_averages d USING(department)
WHERE e.salary>d.avg_salary ORDER BY employee_id;
